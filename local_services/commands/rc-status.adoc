=== rc-status
Mostrar estado de servicios en sistemas con OpenRC.

:category: local_services
[.metadata]
{
  "description": "Mostrar estado de servicios en sistemas con OpenRC.",
  "usage": "rc-status [OPCIONES]",
  "options": {
    "-s, --services": "Mostrar solo servicios activos",
    "-l, --list": "Mostrar runlevels y dependencias",
    "--crashed": "Mostrar servicios caídos",
    "--stopped": "Mostrar servicios detenidos",
    "--inactive": "Mostrar servicios inactivos",
    "-a, --all": "Mostrar todos los servicios instalados",
    "--deps": "Mostrar árbol de dependencias",
    "--list-deps servicio": "Mostrar dependencias específicas de un servicio",
    "--order": "Mostrar orden de inicio de servicios",
    "--time": "Mostrar tiempos de inicio de servicios",
    "--verbose": "Mostrar información detallada de servicios",
    "--full": "Mostrar nombre completo de servicios"
  }
}

[source,bash]
----
# Ver estado general de servicios
rc-status
----

-Estado general:: Vista completa del sistema

[source,bash]
----
# Ver todo con estados
rc-status

# Ver solo servicios activos
rc-status -s

# Ver runlevels y dependencias
rc-status -l
----

-Servicios caídos:: Detección de problemas

[source,bash]
----
# Mostrar servicios caídos (crashed)
rc-status --crashed

# Ver servicios detenidos
rc-status --stopped

# Ver inactivos
rc-status --inactive
----

-Listar todo:: Visión completa

[source,bash]
----
# Ver todos los servicios instalados
rc-status -a

# Ver servicios por runlevel
rc-status runlevel
rc-status boot

# Formato condensado
rc-status -c
----

-Dependencias:: Relaciones entre servicios

[source,bash]
----
# Ver árbol de dependencias
rc-status --deps

# Ver dependencias específicas
rc-status --list-deps servicio

# Ver orden de inicio
rc-status --order
----

-Info detallada:: Más información

[source,bash]
----
# Ver tiempos de inicio
rc-status --time

# Ver con formato extendido
rc-status --verbose

# Ver nombre completo del servicio
rc-status --full
----

-Sintaxis simple:: Formas de uso

[source,bash]
----
# Ver runlevel actual
rc-status

# Ver servicios de boot
rc-status boot

# Ver todo junto
rc-status -a
----
