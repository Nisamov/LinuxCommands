=== ldapadd
Inserta nuevas entradas en el directorio LDAP desde un archivo LDIF.

:category: host_shared_storage
[.metadata]
{
  "description": "Variante de ldapmodify que está configurada por defecto para añadir nuevas entradas.",
  "usage": "ldapadd -H ldap://localhost -x -D 'cn=admin,dc=empresa,dc=com' -W -f usuarios.ldif",
  "options": {
    "-f": "-f (Ruta al archivo LDIF con los datos)",
    "-x": "-x (Simple auth)",
    "-D": "-D (Bind DN)",
    "-W": "-W (Contraseña)",
    "-c": "-c (Continuar en caso de error)",
    "-n": "-n (Simulación sin ejecutar cambios)",
    "-H": "-H (URI del servidor LDAP)"
  }
}

[source,bash]
----
# Añadir un nuevo usuario desde un archivo .ldif
ldapadd -x -D "cn=admin,dc=ejemplo,dc=com" -W -f nuevo_usuario.ldif
----

archivo:: `-f` Ruta al archivo LDIF con los datos

[source,bash]
----
# Especificar el archivo que contiene las entradas a insertar
ldapadd -f datos_importar.ldif
----

identidad:: `-x -D -W` Métodos de autenticación (Simple, DN y Contraseña)

[source,bash]
----
# Autenticarse como administrador solicitando la contraseña por terminal
ldapadd -x -D "cn=admin,dc=ejemplo,dc=com" -W
----

control:: `-c` Continuar el proceso aunque ocurran errores

[source,bash]
----
# Seguir procesando el archivo LDIF si una entrada ya existe o falla
ldapadd -c -f carga_masiva.ldif
----

dry-run:: `-n` Simular la operación para verificar errores

[source,bash]
----
# Probar la inserción sin realizar cambios reales en el directorio
ldapadd -n -f prueba.ldif
----

servidor:: `-H` Especificar la URI del servidor LDAP

[source,bash]
----
# Conectarse a un servidor específico por IP o nombre de host
ldapadd -H ldap://192.168.1.50 -f usuarios.ldif
----
